#include <stdio.h>
#include <string.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "esp_wifi.h"
#include "esp_event.h"
#include "esp_netif.h"
#include "esp_log.h"
#include "nvs_flash.h"
#include "tcp_server.h"
#include "udp_server.h"
#include "esp_wifi_types.h" // 添加此头文件以解决wifi_protocol_t未定义问题

#define PORT 3333

#define TAG "SERVER"
#define MAX_CLIENTS 5


static void wifi_init_softap(void)
{
    // 初始化 NVS
    ESP_ERROR_CHECK(nvs_flash_init());

    // 初始化网络接口
    ESP_ERROR_CHECK(esp_netif_init());

    // 创建默认事件循环
    ESP_ERROR_CHECK(esp_event_loop_create_default());

    // 创建默认的 Wi-Fi AP 网络接口
    esp_netif_t *ap_netif = esp_netif_create_default_wifi_ap();
    if (ap_netif == NULL) {
        ESP_LOGE(TAG, "Failed to create default Wi-Fi AP netif");
        return;
    }

    // 初始化 Wi-Fi 驱动程序
    wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&cfg));

    // 配置 AP 的 Wi-Fi 参数
    wifi_config_t wifi_config = {
        .ap = {
            .ssid = "ESP32C6_AP",              // 设置 AP 的 SSID
            .password = "123456789",           // 设置 AP 的密码
            .max_connection = MAX_CLIENTS,     // 设置最大连接数
            .channel = 6,                      // 设置信道
            .authmode = WIFI_AUTH_WPA2_PSK,    // 设置认证模式为 WPA2_PSK
        },
    };

    // 设置 Wi-Fi 模式为 AP 模式
    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_AP));

    // 配置 AP 的 Wi-Fi 参数
    ESP_ERROR_CHECK(esp_wifi_set_config(WIFI_IF_AP, &wifi_config));

    // 启用 Wi-Fi 6 (802.11ax) 协议
    ESP_ERROR_CHECK(esp_wifi_set_protocol(WIFI_IF_AP, WIFI_PROTOCOL_11B | WIFI_PROTOCOL_11G | WIFI_PROTOCOL_11N | WIFI_PROTOCOL_11AX));

    // 启用 Wi-Fi
    ESP_ERROR_CHECK(esp_wifi_start());

    // 启动 DHCP 服务
    ESP_ERROR_CHECK(esp_netif_dhcps_start(ap_netif));

    // 获取并打印 AP 的 IP 地址
    esp_netif_ip_info_t ip_info;
    if (esp_netif_get_ip_info(ap_netif, &ip_info) == ESP_OK) {
        ESP_LOGI(TAG, "WiFi6 AP IP: " IPSTR, IP2STR(&ip_info.ip));
    } else {
        ESP_LOGE(TAG, "Failed to get AP IP address");
    }
}

void app_main(void)
{

    ESP_ERROR_CHECK(nvs_flash_init());

    // Initialize WiFi
    wifi_init_softap();

    // Start TCP server
    tcp_server_start();
}
